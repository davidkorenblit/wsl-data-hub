<script>
  function switchAnarchyMode(mode) {
    const viewAttack = document.getElementById('view-attack');
    const viewDefense = document.getElementById('view-defense');
    const btnAttack = document.getElementById('btn-mode-attack');
    const btnDefense = document.getElementById('btn-mode-defense');
    
    if (mode === 'attack') {
      viewAttack.classList.remove('hidden');
      viewDefense.classList.add('hidden');
      btnAttack.className = "px-3.5 py-1.5 rounded-lg text-xs sm:text-sm font-bold transition-all bg-white dark:bg-neutral-900 text-blue-600 dark:text-blue-400 shadow-sm";
      btnDefense.className = "px-3.5 py-1.5 rounded-lg text-xs sm:text-sm font-bold transition-all text-neutral-500 dark:text-neutral-400 hover:text-neutral-900 dark:hover:text-white";
    } else {
      viewAttack.classList.add('hidden');
      viewDefense.classList.remove('hidden');
      btnDefense.className = "px-3.5 py-1.5 rounded-lg text-xs sm:text-sm font-bold transition-all bg-white dark:bg-neutral-900 text-rose-600 dark:text-rose-400 shadow-sm";
      btnAttack.className = "px-3.5 py-1.5 rounded-lg text-xs sm:text-sm font-bold transition-all text-neutral-500 dark:text-neutral-400 hover:text-neutral-900 dark:hover:text-white";
    }
  }
</script>
